import { expect, test } from "bun:test"
import { handleQuote, quoteOrder } from "./service"

test("quotes quantities in integer cents", () => {
  expect(quoteOrder([
    { sku: "coffee", quantity: 2, unitPriceCents: 425 },
    { sku: "tea", quantity: 3, unitPriceCents: 175 },
  ])).toBe(1375)
  expect(quoteOrder([])).toBe(0)
})

test("rejects invalid quantities and prices", () => {
  for (const quantity of [-1, 0, 1.5, NaN]) {
    expect(() => quoteOrder([{ sku: "coffee", quantity, unitPriceCents: 425 }])).toThrow()
  }
  for (const unitPriceCents of [-1, 0.5, Infinity]) {
    expect(() => quoteOrder([{ sku: "coffee", quantity: 1, unitPriceCents }])).toThrow()
  }
})

test("serves an order quote and reports invalid input", async () => {
  const request = (quantity: number) => new Request("https://orders.test/quote", {
    method: "POST",
    body: JSON.stringify([{ sku: "coffee", quantity, unitPriceCents: 425 }]),
  })
  const response = await handleQuote(request(2))
  expect(response.status).toBe(200)
  expect(await response.json()).toEqual({ totalCents: 850 })
  expect((await handleQuote(request(-1))).status).toBe(400)
})
